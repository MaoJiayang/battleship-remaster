📋 命令行方式启动参数搜索
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# 快速测试（2x2 网格，10场/对）
node run.js --test

# 快速搜索（3x3 网格，20场/对）
node run.js --preset quick

# 默认搜索（5x5 网格，30场/对）
node run.js

# 完整搜索（细粒度）
node run.js --preset full

# 自定义参数搜索
node run.js --alpha-min 0.3 --alpha-max 0.7 --alpha-step 0.1 --risk-min 0 --risk-max 0.4 --risk-step 0.1 --games 50 --workers 8

# 保存结果到文件
node run.js --preset quick --output result.json

🌐 Web可视化界面方式
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 进入 web 目录
   cd tools/grid-search/web

2. 首次使用需安装依赖
   npm install express cors

3. 启动服务（默认端口3000）
   node server.js

4. 自定义端口启动
   node server.js 4000
   或
   set PORT=4000 && node server.js

5. 浏览器访问
   http://localhost:3000/

6. 在网页上配置参数，点击"启动搜索"即可实时查看进度和榜单

📊 参数说明
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

--alpha-min <n>      Alpha 最小值（信息增益权重，范围 0-1）
--alpha-max <n>      Alpha 最大值
--alpha-step <n>     Alpha 步长

--risk-min <n>       Risk 最小值（风险规避权重，范围 0-1）
--risk-max <n>       Risk 最大值
--risk-step <n>      Risk 步长

--games <n>          每对配置对战次数（越多越准确但耗时越长）
--workers <n>        Worker 线程数（默认为 CPU 核心数）

--output <file>      输出结果到 JSON 文件
-o <file>            同上

💡 使用建议
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

• 首次使用建议先运行 --test 模式验证环境
• 粗调参数用 --preset quick，精调用 --preset default 或自定义
• Worker 数量建议设置为 CPU 核心数，过多反而降低效率
• 每对对战次数至少 20 场，建议 30-50 场以获得稳定结果
• Web 界面适合需要实时监控和中途调整的场景
• 命令行方式适合批量后台运行和脚本自动化
